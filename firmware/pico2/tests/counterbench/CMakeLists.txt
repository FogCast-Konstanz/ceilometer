cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2)
if(NOT PICO_SDK_PATH)
  set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(counterbench C CXX ASM)
pico_sdk_init()
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_executable(counterbench main.c ${ROOT}/src/sipm_counter.c ${ROOT}/src/test_pulse.c)
target_include_directories(counterbench PRIVATE ${ROOT}/include)
pico_generate_pio_header(counterbench ${ROOT}/src/test_pulse.pio)
target_link_libraries(counterbench pico_stdlib hardware_pwm hardware_pio hardware_dma hardware_clocks hardware_vreg hardware_watchdog pico_unique_id)
pico_enable_stdio_usb(counterbench 1)
pico_enable_stdio_uart(counterbench 0)
pico_set_binary_type(counterbench copy_to_ram)
pico_add_extra_outputs(counterbench)
